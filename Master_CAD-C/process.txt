#This pipeline processes cad-c data from nanopore to break up fastq reads into individual nucleosomes, while retaining read ID of interacting nucs. 
# use conda to install last if not working: conda install -c "bioconda/label/cf201901" last 

lastdb Oligo_index/DNABarcode.txt Oligo_index/DNABarcode.txt

# uses file DNABarcode in Oligo_index folder 
#>cad_C1
#GTCGCCTACCTACGCGCGTAGGTAGGCGAC
#>Nano_Y
#CCTGTACTTCGTTCAGTTACGTATTGCT


cat fastq/*.fastq.gz > fastq/allreads.fastq.gz
gunzip fastq/allreads.fastq.gz
		

# lastal needs fastq file in Fastq folder
# maf_bcs split parses maf file from last 


cat Fastq/*.fastq | parallel --pipe  --recstart '\n@' -N10000  lastal -Q 1 Oligo_index/DNABarcode.txt  > out

perl SCRIPTS/maf_bcsplit_IW.pl out | grep + > maf_out_plus.txt



# sort file according to read ID and then by the position of linker in column 4; then use awk to extract read ID, linker start, linker end and read length

sort  -t',' -k1,1 -k4,4n  maf_out_plus.txt | awk -F, -v OFS='\t' '{print $1,$4,$5,$7}'> maf_linkers.txt



#insert extra line so that read end is captured in data (no linkers are found at ends) this compares each line, if subsequent read ID is different it will print read ID and read end position. This will be appended to end of input file with >>

mawk 'id != $1 {print id,pv2,pv2,pv2} {id=$1; pv2=$4}' maf_linkers.txt | sed 's/ /\t/g'  >> maf_linkers1.txt


cat maf_linkers.txt maf_linkers1.txt > maf_linkers2.txt

#sort the file again

sort -k1,1 -k2,2n  maf_linkers2.txt > sortedMAF_linkers.txt


#use awk to check if subsequent lines have same read ID; if they do, the linker end position of top line is subtracted from the linker beginning position of bottom line. Insert sizes are selected as between 13 and 1000nt. 

mawk 'id==$1 {print id,pv,$2,$2-pv} {id=$1; pv=$3}' sortedMAF_linkers.txt | mawk '$4 > 12 && $4 < 1001' | sed 's/ /\t/g' > maf_nucs.txt


#use awk to produce a freq table of insert sizes (between linkers)

mawk -F '\t' '{freq[$4]++} END{for (i in freq) print i, freq[i]}' maf_nucs.txt

sort -t'\t' -k1,1n -k2,1n  maf_nucs.txt > insert_table.txt


#use awk to make a GTF table for use with seqkit

mawk -v OFS='\t' '{print $1,"cad_C","nuc",$2,$3,$6=".",$7="+",$8=".",$9="."}' maf_nucs.txt > GTF_maf_nucs.txt

#Make frequency table of number of inserts (e.g. nucs) per read
cut -f1 <  GTF_maf_nucs.txt | uniq -c | awk -F ' ' '{freq[$1]++} END{for (i in freq) print i, freq[i]}'| sort -t'\t' -k1,1n -k2,1n  > read_ID_countsq.txt


# split file int 5 parts as it's too big to run in seqkit

split -n 5 GTF_maf_nucs.txt

seqkit subseq --gtf xaa fastq/allreads.fastq > Maf_outa.fastq
seqkit subseq --gtf xab Fastq/allreads.fastq > Maf_outb.fastq
seqkit subseq --gtf xac Fastq/allreads.fastq > Maf_outc.fastq
seqkit subseq --gtf xad Fastq/allreads.fastq > Maf_outd.fastq
seqkit subseq --gtf xae Fastq/allreads.fastq > Maf_oute.fastq


cat Maf_out*.fastq > Maf_nucs.fastq


#map data using minimap2

minimap2 -amap-ont --secondary=no --sam-hit-only -t20 sacCer3.fa Maf_nucs.fastq > alignment.sam

#convert to .bed file select for unique reads

#sam2bed -m 48G -R < alignment.sam | awk '$5 >40' | sed 's/_/\t/g' | parsort -k4,4 -k1,1 -k5,5n  > alignment.bed

#or do this if the file is too large: 
samtools view -@16 -b -S alignment.sam > file.bam
bamtobed -i file.bam > alignment.bed 
cat alignment.bed | awk '$5 >40' | sed 's/_/\t/g' | sort -k4,4 -k1,1 -k5,5n  > alignment_q40.bed


#this removes reads from hairpins, line checks that two consecutiove nucs in the same read have similar coordiates (+- 5bp either start or end) if they do then thr prog removes the duplicated nucs to teh end of the nanopore read.  
mawk -v OFS='\t' '((id>=($2-5) && id<=($2+5)) && rd==$4) || ((id2>=($3-5) && id2<=($2+5)) && rd==$4) {cond=0} (rd!=$4) {cond=1} {id=$2; id2=$3; rd=$4} { if (cond) print $1,$2,$3,$4,$5,$6,$7;} '  alignment_q40.bed > alignment_dedup.bed


awk '{gsub(/-/, "_", $4); print $1 "\t" $2 "\t" $3 "\t" $3-$2 "\t" $2+int(($3-$2)/2) "\t" $7 "\t" $4}' alignment_dedup.bed > read_info.txt


python Readinfobed--pairs--CADC.py

sed "s/'//g" read_info.txt.juicer | awk '{$1=$1; print}' > read_info.txt.sed.juicer 

awk '$2 > $6 {print $5,$6,$7,$8,$1,$2,$3,$4,$9} $2<=$6 {print}' read_info.txt.sed.juicer | parsort -k2,2d -k6,6d > read_info.s.juicer
