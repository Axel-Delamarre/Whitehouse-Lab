# select all overlaps
mawk '$1 == $4 && $8 > 3' pairwise.olp.bed > allolp.bed

#calculate overlap percent:
#
#Prints first 8 columns as-is, then adds 4 calculated columns:
#
#     Column 9 : $3-$2 = length of first region
#     Column 10 : $6-$5 = length of second region  
#     Column 11 : int((($3-$2)+($6-$5))/2) = mean length of the two regions (rounded down)
#     Column 12 : mean length / $8 = ratio of mean length to column 8 (overlap size)

mawk '{print $1 "\t" $2 "\t" $3 "\t" $4 "\t" $5 "\t" $6 "\t" $7 "\t" $8 "\t" $3-$2 "\t" $6-$5 "\t" int((($3-$2)+($6-$5))/2) "\t"  int((($3-$2)+($6-$5))/2)/$8 }' allolp.bed > olp.pct.bed
     

# counterselect agains overlap = mean size:
#Filters rows where: 
#
#     $12 <= 0.98 || $12 >= 1.02 : Ratio is NOT ~1.0 (overlap ≠ mean size) - excludes near-identical overlaps
#     $8 > 15 : Overlap length > 15bp
#     $9 >= 100 && $9 <= 1000 : First region is 100-1000bp
#     $10 >= 100 && $10 <= 1000 : Second region is 100-1000bp
     
mawk '($12 <= 0.98 || $12 >= 1.02 && $8 > 15 && $9 >= 100 && $9 <= 1000 && $10 >= 100 && $10 <= 1000)' olp.pct.bed > olp.pct.filt.bed
# Should probably be:
# mawk '(($12 <= 0.98 || $12 >= 1.02) && $8 > 15 && $9 >= 100 && $9 <= 1000 && $10 >= 100 && $10 <= 1000)' olp.pct.bed > olp.pct.filt.test.bed


# Extract the actual overlap:
#    start = ($2 > $5 ? $2 : $5)  
#         Takes the maximum  of the two start positions
#         If region1 starts at position 100 and region2 starts at 150, overlap starts at 150
#    end = ($3 < $6 ? $3 : $6)  
#         Takes the minimum  of the two end positions
#         If region1 ends at position 500 and region2 ends at 400, overlap ends at 400
#    print $1 "\t" start "\t" end  
#         Outputs: chromosome/contig (column 1), overlap start, overlap end
             
awk '{start = ($2 > $5 ? $2 : $5); end = ($3 < $6 ? $3 : $6); print $1 "\t" start "\t" end}' olp.pct.filt.bed > real-olp4.bed
